cmake_minimum_required(VERSION 3.20)
project(DailyStudy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(DailyStudy WIN32 main.cpp)
target_compile_options(DailyStudy PRIVATE /utf-8 /W4)
target_compile_definitions(DailyStudy PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
if(STUDY_PREVIEW)
 target_compile_definitions(DailyStudy PRIVATE STUDY_PREVIEW)
endif()
target_include_directories(DailyStudy PRIVATE "${WEBVIEW2_SDK}/build/native/include")
target_link_libraries(DailyStudy PRIVATE "${WEBVIEW2_SDK}/build/native/x64/WebView2LoaderStatic.lib" ole32 oleaut32 user32 shell32 shlwapi version advapi32)
set_property(TARGET DailyStudy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
